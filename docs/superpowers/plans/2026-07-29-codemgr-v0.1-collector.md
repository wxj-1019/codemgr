# CodeMgr v0.1 — 采集层验证 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建并验证 `codemgr-native` C++ Node-API addon，证明它能用 `NtQuerySystemInformation` 单次调用在 < 20ms 内采集 2000+ 进程——这是整个 C' 方案的头号风险。

**Architecture:** 独立的 Node.js 子包 `codemgr-native/`，用 node-addon-api（C++ wrapper）+ cmake-js（构建）封装 Win32/内核 API。通过 `require('codemgr-native')` 暴露 `processScan()` / `netScan()` / `cpuDelta()` 三个函数。验证通过一个 `bench/` 脚本测量延迟并打印性能报告。本阶段**不写任何 Electron/React UI**。

**Tech Stack:** C++17 / node-addon-api (Node-API) / cmake-js / Windows SDK / NtQuerySystemInformation / GetExtendedTcpTable

**Spec reference:** `docs/superpowers/specs/2026-07-29-codemgr-design.md` §4（采集层设计）、§8 v0.1、§9 风险表

**Go/No-Go 判据（计划完成后必须满足）：**
1. `processScan()` 单次调用对 2000+ 进程 < 20ms
2. `netScan()` 单次调用 < 30ms
3. 连续 60s 高频调用（500ms 间隔）无内存泄漏（RSS 增长 < 10MB）
4. 三项不达标则整个 C' 方案需重新评估（降级到 Pdh/Toolhelp32 或回到 PowerShell）

Create `E:\A_Project\codemgr\codemgr-native\src\process_collector.cpp`:
```cpp
#include "process_collector.h"
#include <windows.h>
#include <winternl.h>
#include <psapi.h>

// NtQuerySystemInformation 的系统调用号声明
// winternl.h 只声明了类号 1，需自行声明完整版本
extern "C" NTSTATUS NTAPI NtQuerySystemInformation(
    ULONG SystemInformationClass,
    PVOID SystemInformation,
    ULONG SystemInformationLength,
    PULONG ReturnLength);

// SystemProcessInformation (类号 5) 的结构
// 字段布局参考 phnt / SI 源码
typedef struct _SYSTEM_PROCESS_INFORMATION {
    ULONG NextEntryOffset;
    ULONG NumberOfThreads;
    LARGE_INTEGER WorkingSetSize;  // 简化：用 WorkingSetPrivateSize 更准但布局复杂
    LARGE_INTEGER HardFaultCount;
    ULONG NumberOfThreadsHighWatermark;
    ULONGLONG CycleTime;
    LARGE_INTEGER CreateTime;
    LARGE_INTEGER UserTime;
    LARGE_INTEGER KernelTime;
    UNICODE_STRING ImageName;
    LONG BasePriority;
    HANDLE UniqueProcessId;
    HANDLE InheritedFromUniqueProcessId;
    ULONG HandleCount;
    ULONG SessionId;
    UNICODE_STRING WindowTitle;
} SYSTEM_PROCESS_INFORMATION;

// 读进程命令行：通过 PEB。NtQueryInformationProcess(ProcessBasicInformation)
// 拿 PEB 地址 → 读 ProcessParameters → 读 CommandLine
static std::wstring ReadProcessCmdline(HANDLE hProcess) {
  // 简化实现：对大多数进程用这个路径
  // 完整实现见 https://github.com/winsiderss/systeminformer 的 pe.c
  // 此处用 WMI 替代方案太慢，先用 QueryFullProcessImageName + 简化
  // TODO(v0.2): 用 NtQueryInformationProcess 读 PEB 拿真实命令行
  return L"";
}

bool CollectAllProcesses(std::vector<ProcessInfoRaw>& out, std::string& errMessage) {
  // 第一次调用拿所需缓冲区大小
  ULONG bufLen = 1024 * 1024;  // 初始 1MB
  std::vector<BYTE> buffer(bufLen);
  ULONG returnLen = 0;
  NTSTATUS status;

  // 循环扩容，直到够用
  while ((status = NtQuerySystemInformation(
              5,  // SystemProcessInformation
              buffer.data(),
              bufLen,
              &returnLen)) == 0xC0000004 /*STATUS_INFO_LENGTH_MISMATCH*/) {
    bufLen *= 2;
    buffer.resize(bufLen);
  }

  if (status != 0) {
    errMessage = "NtQuerySystemInformation failed: " + std::to_string(status);
    return false;
  }

  // 遍历链表
  ULONG offset = 0;
  FILETIME ftNow;
  GetSystemTimeAsFileTime(&ftNow);
  long long nowMs = ((long long)ftNow.dwHighDateTime << 32 | ftNow.dwLowDateTime) / 10000 - 11644473600000LL;

  while (true) {
    auto* p = reinterpret_cast<SYSTEM_PROCESS_INFORMATION*>(buffer.data() + offset);
    ProcessInfoRaw info{};

    info.pid = (ULONG)(ULONG_PTR)p->UniqueProcessId;
    info.ppid = (ULONG)(ULONG_PTR)p->InheritedFromUniqueProcessId;

    // 进程名（UNICODE_STRING → UTF-8）
    if (p->ImageName.Buffer && p->ImageName.Length > 0) {
      int len = WideCharToMultiByte(CP_UTF8, 0,
          p->ImageName.Buffer, p->ImageName.Length / 2,
          nullptr, 0, nullptr, nullptr);
      info.name.resize(len);
      WideCharToMultiByte(CP_UTF8, 0,
          p->ImageName.Buffer, p->ImageName.Length / 2,
          &info.name[0], len, nullptr, nullptr);
    }
    // Idle 进程名为空，补个 "Idle"
    if (info.name.empty() && info.pid == 0) info.name = "Idle";

    // 时间：100ns 单位 → 毫秒
    info.kernelTimeMs = p->KernelTime.QuadPart / 10000;
    info.userTimeMs = p->UserTime.QuadPart / 10000;
    info.workingSetBytes = p->WorkingSetSize.QuadPart;
    info.createTimeMs = p->CreateTime.QuadPart / 10000 - 11644473600000LL;
    info.threadCount = p->NumberOfThreads;
    info.handleCount = p->HandleCount;

    // 命令行：v0.1 暂留空（性能优先验证）
    info.cmdline = "";

    out.push_back(std::move(info));

    if (p->NextEntryOffset == 0) break;
    offset += p->NextEntryOffset;
  }

  return true;
}

Napi::Value ProcessScan(const Napi::CallbackInfo& info) {
  Napi::Env env = info.Env();
  std::vector<ProcessInfoRaw> processes;
  std::string err;
  if (!CollectAllProcesses(processes, err)) {
    Napi::Error::New(env, err).ThrowAsJavaScriptException();
    return env.Null();
  }

  Napi::Array result = Napi::Array::New(env, processes.size());
  for (size_t i = 0; i < processes.size(); i++) {
    Napi::Object obj = Napi::Object::New(env);
    obj.Set("pid", Napi::Number::New(env, processes[i].pid));
    obj.Set("ppid", Napi::Number::New(env, processes[i].ppid));
    obj.Set("name", Napi::String::New(env, processes[i].name));
    obj.Set("cmdline", Napi::String::New(env, processes[i].cmdline));
    obj.Set("kernelTimeMs", Napi::Number::New(env, (double)processes[i].kernelTimeMs));
    obj.Set("userTimeMs", Napi::Number::New(env, (double)processes[i].userTimeMs));
    obj.Set("workingSetBytes", Napi::Number::New(env, (double)processes[i].workingSetBytes));
    obj.Set("createTimeMs", Napi::Number::New(env, (double)processes[i].createTimeMs));
    obj.Set("threadCount", Napi::Number::New(env, processes[i].threadCount));
    obj.Set("handleCount", Napi::Number::New(env, processes[i].handleCount));
    result.Set((uint32_t)i, obj);
  }
  return result;
}
```

> **实现说明（写给执行者）：** `SYSTEM_PROCESS_INFORMATION` 的精确字段偏移在不同 Windows 版本有细微差异。phnt 头文件对此有完整定义。本计划用简化版结构验证性能；若字段读错（如 workingSet 字节数异常），应优先参考 [phnt](https://github.com/winsiderss/systeminformer/tree/master/phnt) 的 `ntexapi.h` 中 `SYSTEM_PROCESS_INFORMATION` 完整定义。命令行采集 deliberately 留到 v0.2，因它需逐进程 `OpenProcess` 读 PEB，会拖慢采集——v0.1 只验证"不打开句柄"的快路径。

```cpp
// TCP 状态码 → 字符串
static const char* TcpStateStr(DWORD state) {
  switch (state) {
    case 1:  return "CLOSED";
    case 2:  return "LISTENING";
    case 3:  return "SYN_SENT";
    case 4:  return "SYN_RCVD";
    case 5:  return "ESTABLISHED";
    case 6:  return "FIN_WAIT1";
    case 7:  return "FIN_WAIT2";
    case 8:  return "CLOSE_WAIT";
    case 9:  return "CLOSING";
    case 10: return "LAST_ACK";
    case 11: return "TIME_WAIT";
    case 12: return "DELETE_TCB";
    default: return "UNKNOWN";
  }
}

// 把 DWORD IP（网络字节序）转字符串
static std::string IpToStr(DWORD ip) {
  in_addr addr;
  addr.S_un.S_addr = ip;
  char buf[INET_ADDRSTRLEN];
  inet_ntop(AF_INET, &addr, buf, sizeof(buf));
  return std::string(buf);
}

bool CollectAllConnections(std::vector<NetConnRaw>& out, std::string& errMessage) {
  // --- IPv4 TCP ---
  DWORD size = 0;
  GetExtendedTcpTable(nullptr, &size, FALSE, AF_INET, TCP_TABLE_OWNER_PID_ALL, 0);
  std::vector<BYTE> buf(size);
  if (GetExtendedTcpTable(buf.data(), &size, FALSE, AF_INET,
                          TCP_TABLE_OWNER_PID_ALL, 0) == NO_ERROR) {
    auto* tbl = reinterpret_cast<MIB_TCPTABLE_OWNER_PID*>(buf.data());
    for (DWORD i = 0; i < tbl->dwNumEntries; i++) {
      auto& e = tbl->table[i];
      NetConnRaw c{};
      c.protocol = "tcp";
      c.localAddr = IpToStr(e.dwLocalAddr);
      c.localPort = ntohs((USHORT)e.dwLocalPort);
      c.remoteAddr = IpToStr(e.dwRemoteAddr);
      c.remotePort = ntohs((USHORT)e.dwRemotePort);
      c.state = TcpStateStr(e.dwState);
      c.pid = e.dwOwningPid;
      out.push_back(std::move(c));
    }
  }

  // --- IPv4 UDP ---
  size = 0;
  GetExtendedUdpTable(nullptr, &size, FALSE, AF_INET, UDP_TABLE_OWNER_PID, 0);
  buf.assign(size, 0);
  if (GetExtendedUdpTable(buf.data(), &size, FALSE, AF_INET,
                          UDP_TABLE_OWNER_PID, 0) == NO_ERROR) {
    auto* tbl = reinterpret_cast<MIB_UDPTABLE_OWNER_PID*>(buf.data());
    for (DWORD i = 0; i < tbl->dwNumEntries; i++) {
      auto& e = tbl->table[i];
      NetConnRaw c{};
      c.protocol = "udp";
      c.localAddr = IpToStr(e.dwLocalAddr);
      c.localPort = ntohs((USHORT)e.dwLocalPort);
      c.remoteAddr = "*";
      c.remotePort = 0;
      c.state = "-";
      c.pid = e.dwOwningPid;
      out.push_back(std::move(c));
    }
  }

  // v0.1 暂不做 IPv6（后续按需加 AF_INET6）
  return true;
}

Napi::Value NetScan(const Napi::CallbackInfo& info) {
  Napi::Env env = info.Env();
  std::vector<NetConnRaw> conns;
  std::string err;
  if (!CollectAllConnections(conns, err)) {
    Napi::Error::New(env, err).ThrowAsJavaScriptException();
    return env.Null();
  }

  Napi::Array result = Napi::Array::New(env, conns.size());
  for (size_t i = 0; i < conns.size(); i++) {
    Napi::Object obj = Napi::Object::New(env);
    obj.Set("protocol", Napi::String::New(env, conns[i].protocol));
    obj.Set("localAddr", Napi::String::New(env, conns[i].localAddr));
    obj.Set("localPort", Napi::Number::New(env, conns[i].localPort));
    obj.Set("remoteAddr", Napi::String::New(env, conns[i].remoteAddr));
    obj.Set("remotePort", Napi::Number::New(env, conns[i].remotePort));
    obj.Set("state", Napi::String::New(env, conns[i].state));
    obj.Set("pid", Napi::Number::New(env, conns[i].pid));
    // processName 留空，由 TS 层用 processScan 结果填充
    obj.Set("processName", Napi::String::New(env, ""));

  }
```

```cpp
#include "cpu_tracker.h"

#include <windows.h>

std::unordered_map<ULONG, CpuTracker::Sample> CpuTracker::last_;

Napi::Value CpuTracker::CpuDelta(const Napi::CallbackInfo& info) {
  Napi::Env env = info.Env();

  // 拿当前进程列表
  std::vector<ProcessInfoRaw> procs;
  std::string err;
  if (!CollectAllProcesses(procs, err)) {
    Napi::Error::New(env, err).ThrowAsJavaScriptException();
    return env.Null();
  }

  // 当前时间（毫秒）
  FILETIME ft;
  GetSystemTimeAsFileTime(&ft);
  long long nowMs = ((long long)ft.dwHighDateTime << 32 | ft.dwLowDateTime) / 10000;

  Napi::Array result = Napi::Array::New(env, procs.size());
  uint32_t outIdx = 0;

  for (const auto& p : procs) {
    auto it = last_.find(p.pid);
    double cpuPercent = 0.0;

    if (it != last_.end()) {
      long long dt = nowMs - it->second.timeMs;
      if (dt > 0) {
        long long cpuMs = (p.kernelTimeMs - it->second.kernelMs)
                        + (p.userTimeMs - it->second.userMs);
        // CPU% = 进程占用CPU时间 / 实际流逝时间 * 100（相对单核）
        cpuPercent = (double)cpuMs / dt * 100.0;
        if (cpuPercent < 0) cpuPercent = 0;
        if (cpuPercent > 100) cpuPercent = 100;
      }
    }

    Napi::Object obj = Napi::Object::New(env);
    obj.Set("pid", Napi::Number::New(env, (double)p.pid));
    obj.Set("cpuPercent", Napi::Number::New(env, cpuPercent));
    result.Set(outIdx++, obj);

    // 更新快照
    last_[p.pid] = {p.kernelTimeMs, p.userTimeMs, nowMs};
  }

  // 清理已退出的进程（避免 map 无限增长）
  // 简化策略：若 map 比当前进程列表大很多，整体重建
  if (last_.size() > procs.size() * 2) {
    last_.clear();
    for (const auto& p : procs) {
      last_[p.pid] = {p.kernelTimeMs, p.userTimeMs, nowMs};
    }
  }

  return result;
}
```

```cpp
#include "process_ops.h"

Napi::Value KillProcess(const Napi::CallbackInfo& info) {
  Napi::Env env = info.Env();
  if (info.Length() < 1 || !info[0].IsNumber()) {
    Napi::TypeError::New(env, "expected pid:number").ThrowAsJavaScriptException();
    return env.Null();
  }
  DWORD pid = (DWORD)info[0].As<Napi::Number>().Int32Value();

  HANDLE h = OpenProcess(PROCESS_TERMINATE, FALSE, pid);
  if (!h) {
    return Napi::Boolean::New(env, false);  // 失败：权限不足或进程已退出
  }
  BOOL ok = TerminateProcess(h, 1);
  CloseHandle(h);
  return Napi::Boolean::New(env, ok != 0);
}

Napi::Value KillByName(const Napi::CallbackInfo& info) {
  Napi::Env env = info.Env();
  if (info.Length() < 1 || !info[0].IsString()) {
    Napi::TypeError::New(env, "expected name:string").ThrowAsJavaScriptException();
    return env.Null();
  }
  std::string target = info[0].As<Napi::String>().Utf8Value();

  // 采集后匹配名
  std::vector<ProcessInfoRaw> procs;
  std::string err;
  if (!CollectAllProcesses(procs, err)) {
    Napi::Error::New(env, err).ThrowAsJavaScriptException();
    return env.Null();
  }

  int killed = 0;
  for (const auto& p : procs) {
    // 大小写不敏感比较（Windows 进程名不区分大小写）
    if (_stricmp(p.name.c_str(), target.c_str()) == 0) {
      HANDLE h = OpenProcess(PROCESS_TERMINATE, FALSE, p.pid);
      if (h) {
        if (TerminateProcess(h, 1)) killed++;
        CloseHandle(h);
      }
    }
  }
  return Napi::Number::New(env, killed);
}
```
